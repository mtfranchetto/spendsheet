/** * How far right you have to drag before the swipe means anything. * * A fraction of the row rather than a fixed distance, so it feels the same on * a 320px phone and a tablet — bounded at both ends, because a quarter of a * 900px row is a swipe nobody completes and a quarter of a narrow one is * close enough to a stray finger to fire by accident. */ function swipeThreshold(width) { return Math.min(96, Math.max(60, width * 0.25)); } /** * The distance that decides whether a gesture is a scroll or a swipe. * * Below this nothing is claimed, because the first few pixels of every * gesture are ambiguous and guessing early is what makes a list feel like it * fights you. */ var SWIPE_INTENT = 10; /** * Gestures starting this close to the left edge belong to the browser. * * iOS puts back-navigation there, and a row that swallowed it would trap * someone on a screen. Losing the leftmost 24px of a gesture that has 60px * to travel is the cheaper side of that trade. */ var EDGE_GUARD = 24; /** * Swipe a row right to confirm it. * * The gesture the whole screen was missing: confirming used to be tap → wait * for a sheet → tap Confirm → the sheet closes, three actions and a modal for * the answer that is right most of the time. One transform moves, and there is no * second element whose position has to be kept in step. * * Five things this has to get right, each of which is a real failure: * * **Vertical wins.** The direction is decided once, at `SWIPE_INTENT`, and * a gesture that turns out to be vertical is abandoned for good rather than * re-evaluated — otherwise a diagonal scroll flickers between the two and * the list stutters under your thumb. * * **The browser keeps scrolling.** `touch-action: pan-y` hands vertical * panning back to the compositor, so the list scrolls at 60fps without this * code running at all. Reimplementing scroll here would be both slower and * worse. * * **The left edge is not ours.** See `EDGE_GUARD`. * * **The transform is cleared before anything renders.** `sendDecision` * calls `render()`, which rebuilds `#screens` wholesale — so a row left * mid-transform is not restored, it is destroyed, and the *next* row drawn * into that position inherits nothing. Resetting first keeps the two * independent. * * **The click that follows is suppressed.** A completed drag still ends in * a `click`, which would open the sheet on top of the confirmation that * just happened. */ function swipeToConfirm(button, record) { var wrap = el('div', 'swipe'); var action = el('div', 'swipe-action', '\u{2713}'); action.setAttribute('aria-hidden', 'true'); add(wrap, action); add(wrap, button); var startX = 0, startY = 0, dx = 0; var horizontal = null, tracking = false; var settle = function (animate) { wrap.className = 'swipe' + (animate ? ' springing' : ''); button.style.transform = ''; dx = 0; horizontal = null; tracking = false; }; button.addEventListener('touchstart', function (event) { // Two fingers is a pinch or a scroll, and neither is this. if (event.touches.length !== 1) return settle(false); var touch = event.touches[0]; if (touch.clientX < EDGE_GUARD) return; startX = touch.clientX; startY = touch.clientY; dx = 0; horizontal = null; tracking = true; wrap.className = 'swipe'; }, { passive: true }); button.addEventListener('touchmove', function (event) { if (!tracking) return; var touch = event.touches[0]; var moveX = touch.clientX - startX; var moveY = touch.clientY - startY; if (horizontal === null) { if (Math.abs(moveX) < SWIPE_INTENT && Math.abs(moveY) < SWIPE_INTENT) return; horizontal = Math.abs(moveX) > Math.abs(moveY); // Decided once. A gesture that started vertical stays vertical. if (!horizontal) { tracking = false; return; } } // Claimed only now, once the direction is known — before this point the // browser must stay free to scroll. event.preventDefault(); // Rightwards only. Deleting lives in the sheet, on the one row kind that // can be deleted — see `deleteControl` — deliberately rather than on the // leftward swipe this leaves free: almost every row is synced and cannot // be deleted at all, and a gesture that does nothing on almost every row // teaches you it is broken. dx = Math.max(0, moveX); button.style.transform = 'translateX(' + dx + 'px)'; wrap.className = 'swipe sliding' + (dx >= swipeThreshold(button.offsetWidth) ? ' armed' : ''); }, { passive: false }); button.addEventListener('touchcancel', function () { settle(true); }, { passive: true }); button.addEventListener('touchend', function () { if (!tracking || !horizontal) return settle(false); var fired = dx >= swipeThreshold(button.offsetWidth); // Any completed horizontal drag suppresses its click, not only one that // crossed the line — a drag that fell short is still not a tap. button.swiped = true; settle(true); if (!fired) return; if (record.confirmed) return toast('Already confirmed.'); sendDecision(record, { uid: record.uid, confirmed: true }); }, { passive: true }); return wrap; } /** * One transaction, as a button. * * A button rather than a div with a click handler, because a button is what * a screen reader announces as actionable and what a keyboard reaches with * Tab.
