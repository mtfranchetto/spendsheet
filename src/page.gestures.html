// ------------------------------------------------------------------------- // Pull to refresh // ------------------------------------------------------------------------- /** How far down you have to drag before releasing means anything. */ var PULL_THRESHOLD = 70; /** Past this the indicator stops following your finger, so it cannot run away. */ var PULL_MAX = 110; /** * Drag down at the top of the page to re-read everything. * * `appDashboard` was written for this and had no gesture attached to it for * three phases: it is the one entry point that calls `forgetLedgerCache_()` * before reading, so it is the only way from a phone to say "I do not care * what you have cached, go and look". Everything else the app does is happy * with a six-hour-old ledger, because every *write* bumps the version. * * Two conditions, and both matter: * * **Only from the very top.** `scrollY > 0` means you are scrolling a list * and a downward drag is how you get back up it. Claiming that gesture * would make the page feel like it was fighting you, for a refresh you * never asked for. * * **Only downward, and abandoned the moment it turns horizontal.** The rows * below have their own horizontal gesture, and a pull-to-refresh that ate * the start of a swipe would make confirming a transaction unreliable in * exactly the place you do it most. * * `passive: true` throughout — this never calls `preventDefault`, because at * `scrollY === 0` there is nothing above to scroll to and the browser's own * overscroll is the visual feedback we are decorating rather than replacing. */ function installPullToRefresh() { var node = document.getElementById('pull'); var startY = 0; var startX = 0; var pulling = false; var distance = 0; // Built once and mutated, rather than rebuilt per touchmove. `touchmove` // fires at the screen's refresh rate, and replacing two nodes sixty times a // second is how a gesture starts dropping frames on the device it is for. var pill = el('div', 'pill'); var arrow = el('span', 'arrow', '\u{2193}'); var caption = el('span', null, ''); add(pill, arrow); add(pill, caption); add(node, pill); var show = function (amount, text, armed) { node.hidden = false; caption.textContent = text; node.setAttribute('data-armed', armed ? 'true' : 'false'); node.style.transform = 'translateY(' + (amount - 56) + 'px)'; node.style.opacity = Math.min(1, amount / PULL_THRESHOLD); }; var hide = function () { node.hidden = true; node.style.transform = ''; node.style.opacity = ''; node.setAttribute('data-armed', 'false'); pulling = false; distance = 0; }; document.addEventListener('touchstart', function (event) { if (event.touches.length !== 1) return hide(); if (window.scrollY > 0) return; if (state.refreshing) return; startY = event.touches[0].clientY; startX = event.touches[0].clientX; pulling = true; distance = 0; }, { passive: true }); document.addEventListener('touchmove', function (event) { if (!pulling) return; var moveY = event.touches[0].clientY - startY; var moveX = event.touches[0].clientX - startX; // Upward, sideways, or scrolled away from the top: not this gesture. if (moveY <= 0 || Math.abs(moveX) > Math.abs(moveY) || window.scrollY > 0) { return hide(); } // Damped past the threshold rather than clamped, so it keeps responding // to your finger without the indicator sliding down the whole screen. distance = moveY < PULL_THRESHOLD ? moveY : PULL_THRESHOLD + Math.min(PULL_MAX - PULL_THRESHOLD, (moveY - PULL_THRESHOLD) * 0.4); var armed = distance >= PULL_THRESHOLD; show(distance, armed ? 'Release to refresh' : 'Pull to refresh', armed); }, { passive: true }); document.addEventListener('touchend', function () { if (!pulling) return; var fired = distance >= PULL_THRESHOLD; if (!fired) return hide(); show(PULL_THRESHOLD, 'Refreshing\u{2026}', true); state.refreshing = true; // Everything, not just the dashboard: a pull is the gesture for "this is // stale", and refreshing only the screen behind the indicator would leave // the other two exactly as wrong as they were. invalidate(); loadDashboard(true); refreshCurrentScreen(); // Nothing here returns a promise the gesture can wait on — `loadDashboard` // renders when it lands — so the indicator is dismissed on a timer rather // than pretending to know. Long enough to be legible, short enough not to // sit there after the screen has already redrawn beneath it. setTimeout(function () { state.refreshing = false; hide(); }, 900); }, { passive: true }); document.addEventListener('touchcancel', hide, { passive: true }); }
