function renderDashboard(root) { var open = state.data.open; var data = state.data.overview; if (!open) return add(root, spinner()); add(root, el('h1', null, 'Spendsheet')); (data ? data.health : []).forEach(function (alert) { var banner = el('div', 'banner ' + (alert.kind === 'consent' ? 'warn' : 'bad')); add(banner, el('div', null, alert.subject)); add(banner, el('div', 'muted', alert.detail)); add(root, banner); }); // Net worth, then the accounts it is made of. // // The tiles were briefly first, on the argument that a total is not a thing // you can act on. True, and it turns out to be the wrong end to start from: // the total is what you open the app to see, and the tiles are what you // reach for once it has told you something. Summary then detail, which is // the order the rest of this screen already reads in. var worth = el('div', 'card'); add(worth, el('div', 'muted', 'Net worth')); add(worth, el('div', 'big num', money(open.netWorth.total))); if (open.netWorth.skipped) { add(worth, el('div', 'muted', open.netWorth.skipped + ' account(s) not counted — either out of net worth, or not in ' + open.currency)); } add(root, worth); add(root, el('h2', null, 'Accounts')); if (!open.accounts.length) { add(root, add(el('div', 'card'), el('p', 'muted', 'No accounts yet — connect a bank from the Sheet.'))); } else { var tiles = el('div', 'tiles'); open.accounts.forEach(function (account) { add(tiles, accountTile(account, open.currency)); }); add(root, tiles); } // Everything below here needs the ledger. One spinner in the flow, so the // blocks above do not move when it resolves. if (!data) { var pending = el('div', 'card'); if (state.overviewError) { add(pending, el('div', 'muted', state.overviewError)); add(pending, el('div', 'muted', 'Net worth and balances above are current. Reload to try the rest again.')); } else { add(pending, spinner()); } add(root, pending); return; } // The one block on this screen that had no heading, which made it read as // a caption on the accounts above rather than as the month. It pairs with // "This month, by category" below it — same window, one as three numbers // and one as a breakdown — and saying so is most of what the heading buys. add(root, el('h2', null, 'This month')); var month = el('div', 'card'); add(month, add(el('div', 'row'), el('span', null, 'Spent since ' + shortDate(data.month.from)), el('span', 'num out', money(data.month.expenses)))); add(month, add(el('div', 'row'), el('span', null, 'Earned'), el('span', 'num in', money(data.month.income)))); add(month, add(el('div', 'row'), el('span', null, 'Net'), el('span', 'num ' + (data.month.net < 0 ? 'out' : 'in'), money(data.month.net)))); // Month to date against month to date. A comparison that flatters you on // the 3rd and alarms you on the 30th is worse than none at all. var difference = data.month.expenses - data.month.lastMonthToDate; add(month, add(el('div', 'row'), el('span', 'muted', 'Same point last month'), el('span', 'num muted', money(data.month.lastMonthToDate) + ' (' + (difference >= 0 ? '+' : '') + money(difference) + ')'))); add(root, month); add(root, el('h2', null, 'This month, by category')); add(root, add(el('div', 'card'), categoryBars(data.month.byCategory))); add(root, el('h2', null, 'Needs review')); var review = el('div', 'card'); if (!data.review.length) { add(review, el('p', 'muted', 'Nothing waiting — every transaction is confirmed.')); } data.review.slice(0, 10).forEach(function (record) { add(review, recordRow(record)); }); if (data.review.length > 10) { var more = el('button', 'link', 'See all ' + data.review.length + ' in Records'); more.addEventListener('click', function () { // Changing a filter and then navigating has to *reload*: `go` only // fetches when the screen has nothing on it, which after a first visit // is never — so without this the list arrives unfiltered and looks // like the toggle did nothing. state.filters.records.unconfirmedOnly = true; go('records'); loadRecords(); }); add(review, more); } add(root, review); // Holds last, under the queue, and this is the second position they have // had. // // They were directly under the balances, on the argument that a hold is the // reason a balance is not the number you expected. True, and it put five // rows you cannot act on between the tiles and the month — the owner asked // for them lower, and the ordering that argument produces is better anyway: // everything above is a number or something to do, and this is neither. // It is the footnote it reads as. // // What it deliberately does not claim is that the tiles are wrong by this // amount, because that depends on which balance your bank reports: // `BALANCE_PREFERENCE` prefers `CLBD`, which is booked and excludes holds, // but falls through to `ITAV`, which is available and already nets them. // The claim that holds whatever the bank sent is the one made below — these // rows are not in `transactions`, so nothing computed from the ledger // counts them. var held = data.pending || []; if (held.length) { add(root, el('h2', null, 'Pending')); var holds = el('div', 'card'); var group = el('div', 'daygroup'); held.slice(0, 5).forEach(function (record) { add(group, pendingRow(record)); }); add(holds, group); if (held.length > 5) { var allHolds = el('button', 'link', 'See all ' + held.length + ' in Records'); allHolds.type = 'button'; allHolds.addEventListener('click', function () { // Clearing the account filter is what makes the label true. Records // scopes its own holds to whatever it is filtered to — correctly, and // it is why the dashboard carries its own copy — so landing there // still filtered to one account would show fewer than the number the // button just promised. state.filters.records.account = 'All'; state.filters.records.unconfirmedOnly = false; go('records'); loadRecords(); }); add(holds, allHolds); } add(holds, el('p', 'pending-note', 'Not settled yet, so no total on this screen counts them')); add(root, holds); } // The two round-trip timings that used to print here are gone. They were // instrumentation for the sessions that took the open from ~7s to ~6s, and // they earned their place then; what they are now is a line of diagnostics // at the bottom of a finance app. `describeDashboardCost` measures the same // thing properly, from the menu, part by part, and it always did the half // of it that mattered. Profiling the client is Playwright over CDP against // the live deployment, which is the only vantage point that can see the // ~3,300ms of Google that this line could never include anyway. } /** * An account's state as one character, rather than the word "ok". * * `accountState_` returns a sentence, and for a healthy account that sentence * is "ok" — a word repeated down the screen saying nothing, which is worse * than silence because it costs a line to read before you learn that. The * emoji carries the same fact pre-attentively: you see the one that is not * green without reading anything. * * Cash gets its own mark rather than a green one. A cash account has no * consent and no sync, so `accountState_` gives it severity 0 by definition — * calling that "healthy" would be reporting a check that never ran. */ function accountEmoji(account) { if (isCashAccount(account.name)) return '\u{1F4B5}'; if (account.severity === 2) return '\u{1F534}'; if (account.severity === 1) return '\u{1F7E0}'; return '\u{1F7E2}'; } /** The server already knows which accounts take cash entries; it sends the list. */ function isCashAccount(name) { return state.lists.cashAccounts.indexOf(name) !== -1; } /** * One account, as something you can press. * * The tile is a `div` and the two things you can do to it are real buttons * inside it. Pressing the face means "show me what is in this account", which * is the only question a balance ever prompts; a cash account gets a second, * smaller action because it is the only kind you can add to by hand. * * Nesting a button inside a button would be neither — a keyboard reaches one * of them and a screen reader announces something that does not exist. * * **The whole tile is the target, and the listener is on the tile for that * reason rather than on the face.** A button is only as big as its box, and * this one is not the tile: the tile's 12/14px padding is outside it, a grid * row stretches every tile to match its tallest neighbour and the face does * not grow with it, and "+ Add cash" is `align-self: flex-start`, so the * space to its right belongs to nothing at all. Four dead strips around a * control whose entire job is to be tapped with a thumb. * * Delegating from the tile fixes all four at once and keeps the button, which * is what a keyboard tabs to and what a screen reader announces — its own * click bubbles here like any other. The secondary actions are the one * exception, and they are excluded by asking where the click landed rather * than by `stopPropagation` in each of them: this way a third action added * later is a `button.link` and is handled, instead of being a tile that * navigates when you meant to press it. */ function accountTile(account, currency) { var tile = el('div', 'tile' + (account.severity ? ' sev' + account.severity : '')); var face = el('button', 'face'); face.type = 'button'; var who = el('div', 'who'); add(who, el('span', null, accountEmoji(account))); add(who, add(el('div', 'ellipsis grow'), document.createTextNode(account.name))); add(face, who); add(face, el('div', 'amount num', account.balance === null ? '—' : money(account.balance) + (account.currency === currency ? '' : ' ' + account.currency))); // Only when it says something. "ok" and "—" are the two answers worth no line. if (account.severity) add(face, el('div', 'muted note', account.sync)); face.setAttribute('aria-label', 'Show records for ' + account.name + (account.severity ? '. ' + account.sync : '')); add(tile, face); tile.addEventListener('click', function (event) { // A secondary action inside the tile is itself and not the tile. if (event.target.closest && event.target.closest('button.link')) return; // Same reload the review queue's "see all" needs: `go` only fetches a // screen that has nothing on it, so after a first visit the filter would // change and the list would not. state.filters.records.account = account.name; state.filters.records.unconfirmedOnly = false; go('records'); loadRecords(); }); if (isCashAccount(account.name)) { var entry = el('button', 'link', '+ Add cash'); entry.type = 'button'; entry.addEventListener('click', function () { state.addAccount = account.name; go('add'); }); add(tile, entry); } // An asset is a wallet you top up by stating the total rather than the // change: you read 14,500 off a broker's app, you do not compute the €1,500 // it moved by. The screen does that subtraction — see `renderSetValue`. if (assetAccount(account.name)) { var revalue = el('button', 'link', '\u{2195} Update balance'); revalue.type = 'button'; revalue.addEventListener('click', function () { state.addAccount = account.name; go('revalue'); }); add(tile, revalue); } return tile; } /** The asset entry for a name, or null. Carries its balance — see `assetAccountList_`. */ function assetAccount(name) { return (state.lists.assetAccounts || []).filter(function (entry) { return entry.name === name; })[0] || null; }
