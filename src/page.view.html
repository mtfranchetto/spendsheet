// ------------------------------------------------------------------------- // Screens // ------------------------------------------------------------------------- var SCREENS = [ { id: 'dashboard', label: 'Home', glyph: '\u{1F3E0}', render: renderDashboard }, // The tab bar draws in *this* order — it filters `SCREENS` by `NAV` // membership rather than mapping over `NAV` — so the two lists have to // agree on order as well as on membership, and this is the one that shows. { id: 'analytics', label: 'Charts', glyph: '\u{1F4CA}', render: renderAnalytics }, { id: 'records', label: 'Records', glyph: '\u{1F4C4}', render: renderRecords }, { id: 'add', label: 'Add cash', glyph: '\u{2795}', render: renderAdd }, // Reached from an asset tile only, like `add` from a cash one. { id: 'revalue', label: 'Update balance', glyph: '\u{2195}', render: renderSetValue } ]; /** * Which screens the tab bar offers, which is no longer all of them. * * Adding cash used to be a tab, and a tab is a claim about how often you * reach for something. You add cash from the account you spent it out of — * which is a tile on the dashboard, and which also answers the "into which * account" question the form would otherwise open by asking. So the screen * stays and the tab goes: `go('add')` still reaches it, from the one place * where it is already in context. */ var NAV = ['dashboard', 'analytics', 'records']; /** * The query string, spelled once. `WEBAPP_VIEW_PARAM` and friends in * `20_webapp.gs` are the other half, and the two must agree — this writes * what that reads, with nothing in between to catch a drift. * * **This is not every filter Records has.** The three a drill sets — * `spendOnly`, `category`, `subcategory` — are deliberately absent, so a * refresh after tapping a number on Charts comes back to Records without the * drill. That loses state rather than misreporting it: the chips are drawn * from the same filters the query used, so they disappear together and the * screen never claims a narrowing it did not apply. Putting them here means * three more params, their mirror in `20_webapp.gs`, and `applyUrlFilters` * validating a category name against a list — worth doing, not worth doing * silently as part of a merge. */ var VIEW_PARAMS = { view: 'v', period: 'p', from: 'from', to: 'to', account: 'a', label: 'l', search: 'q', unconfirmed: 'u' }; /** * What the URL asked for, read once from the document the server built. * * The screen is validated here as well as on the server, and not because the * server's answer is doubted: this reads a DOM element, and the honest * failures are a renamed placeholder printing nothing and a template that was * never filled. Either one produces a `state.screen` that `render()` cannot * find a screen for, which is a blank page rather than a wrong tab. * * The filters are taken as they come. They are bounded server-side and they * are checked against the real lists in `applyUrlFilters`, which is the only * place those lists exist. */ function readView() { var fallback = { screen: NAV[0], filters: {} }; try { var node = document.getElementById('view'); var parsed = JSON.parse(String(node ? node.textContent : '').trim() || 'null'); if (!parsed || typeof parsed !== 'object') return fallback; return { screen: NAV.indexOf(parsed.screen) === -1 ? NAV[0] : parsed.screen, filters: parsed.filters && typeof parsed.filters === 'object' ? parsed.filters : {} }; } catch (e) { return fallback; } } /** Read at parse, applied at boot, and emptied by `applyUrlFilters` after. */ var opened = readView(); /** * Put the screen and its filters on the URL, so a refresh comes back to them. * * This page is inside Apps Script's sandboxed iframe, and `window.location` * here is a `userCodeAppPanel` URL that nothing ever reloads — a refresh * reloads the *parent*, at `/exec`. `google.script.history` is the one API * that reaches it, and it is why this is a query string rather than a hash: a * parameter comes back through `doGet` and the page is *served* knowing the * answer, where a hash would only be readable after the page had already * drawn the wrong thing. * * `replace` rather than `push`. What was asked for is that a refresh returns * you where you were, and pushing an entry per filter change would make the * phone's back gesture walk every period you had tried instead of leaving the * app — a different feature, and one that needs `setChangeHandler` to be * correct rather than merely present. * * **Only what differs from the default is written.** A URL that spelled out * every filter would freeze today's defaults into every bookmark, so the day * `appDefaultFilters_` changes its mind about what Records opens on, nothing * anyone had saved would hear about it. It also keeps a fresh open at a bare * `?v=dashboard`, which is the address that gets added to a home screen. * * Only the three tab-bar screens, matching `WEBAPP_SCREENS`. `add` and * `revalue` are opened *from an account tile* and carry that account in * memory only, so naming one would restore a form that had forgotten what it * was for. Leaving the URL alone means a refresh from either lands on the * screen you opened it from, which is the way back they already offer. * * Wrapped, because a URL that fails to update is a worse refresh and not a * broken app — `google.script.history` exists only in a deployed web app, so * this is silently a no-op anywhere else. */ function rememberView() { if (NAV.indexOf(state.screen) === -1) return; var params = {}; params[VIEW_PARAMS.view] = state.screen; var filters = state.filters[state.screen]; if (filters) { var period = filters.period || {}; var id = periodId(period); if (id) params[VIEW_PARAMS.period] = id; // Both ends, and only for `custom` — they are the only part of a period // spec that is not recoverable from its id. if (period.kind === 'custom') { if (period.from) params[VIEW_PARAMS.from] = period.from; if (period.to) params[VIEW_PARAMS.to] = period.to; } // Repeated rather than joined: the account filter is a set, and a // separator is a bet that no account is ever named with it in it. var accounts = chosenAccounts(filters); if (accounts.length) params[VIEW_PARAMS.account] = accounts; if (filters.label && filters.label !== 'All') { params[VIEW_PARAMS.label] = filters.label; } if (filters.search) params[VIEW_PARAMS.search] = filters.search; if (filters.unconfirmedOnly) params[VIEW_PARAMS.unconfirmed] = '1'; } try { if (window.google && google.script && google.script.history) { google.script.history.replace(null, params); } } catch (e) { // Nothing to say and nowhere to say it — the screen is already drawn. } } /** * The account filter as a list, whatever shape it is in. * * It is `'All'`, or one name, or an array of them — `accountChips` writes all * three and has since it replaced the dropdown. Both the URL writer and the * chips need the same normalisation, so it is spelled here once. */ function chosenAccounts(filters) { if (Array.isArray(filters.account)) return filters.account.slice(); return filters.account && filters.account !== 'All' ? [filters.account] : []; } /** * The URL's filters, over the defaults, once there are lists to resolve them. * * Called from `applyOverview` and nowhere else, because that is the moment * `periods` exists: a period id is `mtd` or `year:2024`, and only the list * the server sent says which kind and year that is. Anything it cannot * resolve is left at the default rather than guessed at — a bookmark naming a * period that no longer exists should open on this month, not on a window * assembled out of half an id. * * Returns whether it changed anything, compared through `signature` so that a * URL spelling out the default costs no extra fetch. `opened.filters` is * emptied on the way out: this is the URL *the document was served for*, and * applying it twice would drag you back to it on your second visit to a * screen you had since filtered differently. */ function applyUrlFilters() { var screen = opened.screen; var asked = opened.filters || {}; var filters = state.filters[screen]; opened.filters = {}; if (!filters) return false; var before = signature(currentFilters(screen)); if (asked.period === 'custom') { // Both ends or neither. One of them alone resolves to a window whose // other edge is today, which is not the range that was bookmarked. if (asked.from && asked.to) { filters.period = { kind: 'custom', from: asked.from, to: asked.to }; } } else if (/^month:\d{4}-\d{2}$/.test(String(asked.period || ''))) { // Its own branch, like `custom`, because a month is the other id the // list cannot resolve: `periods` carries one generic `month` entry and // this names a particular one. Read here rather than looked up, and the // pattern is checked first — `month:` followed by anything else is a // hand-edited URL, and the default is a better answer than a window // assembled out of a NaN. var parts = String(asked.period).slice(6).split('-'); filters.period = { kind: 'month', year: Number(parts[0]), month: Number(parts[1]) }; } else if (asked.period) { var entry = (state.lists.periods || []).filter(function (option) { return option.id === asked.period; })[0]; // Kept whole, exactly as `filterBar` does: a bare `{kind: 'year'}` with // no year in it resolves to the current year, so 2024 would show 2026. if (entry) filters.period = { kind: entry.kind, year: entry.year }; } if (asked.accounts && asked.accounts.length) { filters.account = asked.accounts.slice(); } if (asked.label) filters.label = asked.label; // Records only, and this is not tidiness. Charts draws its filter bar // without a search box or a review toggle, so either one set here would // narrow every total on the screen with no control saying so and no way to // clear it. if (screen === 'records') { if (asked.search) filters.search = asked.search; if (asked.unconfirmed) filters.unconfirmedOnly = true; } return signature(currentFilters(screen)) !== before; } /** * Draw what has arrived, rather than nothing until everything has. * * The screen used to be one call and one spinner, so the cheapest half of it * — net worth and the tiles, which need no ledger — waited on the most * expensive. Server time had come down from ~6,200ms to ~2,400 and the app * still felt slow, because **what a person experiences is when the first * real content appears, not when the last does.** * * So `open` and `overview` are fetched at the same time and each block draws * as its half lands. `open` is ~800ms and holds the top of the screen; the * rest keeps its own spinner until the ledger arrives, in place rather than * over the whole page, so nothing that is already on screen moves when it * does. */
