Returns where each row was, * because a refusal has to put it back exactly there: appending it would * silently re-sort someone's list under them, and dropping it for good would * hide a transaction that is still unconfirmed on the server. * * **The lists are searched by identity and the same transaction is a separate * object in each**, because they arrive from separate calls. So exactly one * list ever gives up a row here — whichever one you tapped. The other is * marked stale by `invalidate()` and refetched the next time it is asked for, * which is what stops the two disagreeing. * * The queue holds up to `queueSize_()` and the dashboard draws ten of them, so * removing one does reveal the next. That is the behaviour wanted rather than * a leak: the row you acted on leaves, the "see all" count beside it falls, * and the backlog closes up the way an inbox does. */ function dropConfirmed(record) { return dropFromLists(record, [ state.filters.records.unconfirmedOnly ? state.records.rows : null, state.data.overview && state.data.overview.review ]); } /** * The same removal, for a row that is not coming back at all. * * A delete takes it out of `state.records.rows` unconditionally — where * `dropConfirmed` leaves it alone unless the list is filtered to unconfirmed, * because a confirmed row is what an unfiltered list is *for*. There is no * filter under which a deleted row belongs on screen. */ function dropRecord(record) { return dropFromLists(record, [ state.records.rows, state.data.overview && state.data.overview.review ]); } /** * Take the record out of each list that still holds it, remembering where. * * The count follows `state.records.rows` **by identity** rather than by which * caller ran, for the same reason `restoreDropped` does: a refetch that lands * mid-flight replaces that array wholesale, and a decrement aimed at the array * it replaced would be counting a list nobody is drawing. */ function dropFromLists(record, lists) { var dropped = []; (lists || []).forEach(function (rows) { var at = (rows || []).indexOf(record); if (at === -1) return; rows.splice(at, 1); dropped.push({ rows: rows, at: at }); if (rows === state.records.rows && state.records.total > 0) state.records.total--; }); return dropped; } /** * Put back what a refusal did not earn. * * The count follows the array it was counting, by identity rather than by * remembering which branch above ran: a refetch that landed in between * replaces `state.records.rows` wholesale, and restoring into the array it * replaced must not then claim a row on the one now being drawn. */ function restoreDropped(record, dropped) { (dropped || []).forEach(function (entry) { entry.rows.splice(entry.at, 0, record); if (entry.rows === state.records.rows) state.records.total++; }); } /** * A category's kind, from the list the bootstrap already sends. * * Only used to predict what the sheet is about to derive, so the row can be * redrawn before the server answers. `typeFormula()` is the authority and * this is a guess at what it will say — which is why `reloadAfterWrite` * follows every write and replaces the guess with the answer. */ function kindOfCategory(name) { var found = (state.lists.categories.parents || []).filter(function (entry) { return entry.name === name; })[0]; return found ? found.kind : ''; } /** * Replace an optimistic guess with what the server actually says. * * In place rather than by swapping the object: `state.records.rows` holds * these by reference, `dropConfirmed` remembers an index into that array, * and the sheet that is still open closed over this exact record. Assigning a * new object would leave three of those pointing at the old one. * * Keyed off the server's own shape, so a field added to `recordForPage_` * arrives here without a second list to keep in step. * * Draws nothing. A batch patches several rows from one answer and the screen * is redrawn once at the end of it — see `flushDecisions`. */ function patchRecord(record, fresh) { Object.keys(fresh).forEach(function (key) { record[key] = fresh[key]; }); } /** * Decisions waiting for a call to carry them, and whether one is already out. * * **One `appConfirm` in flight at a time, carrying everything swiped since the * last one left.** `appConfirm` has always taken a list — its own comment says * why, at length: a request per tap produces a queue of executions that * resolve out of order, and on Apps Script they queue rather than overlap, so * the fifth swipe's answer arrives seconds after the fifth swipe and the * bridge in front of them eventually drops one with `http connection failure * 0`. The page sent that list one element at a time anyway, because a swipe is * one row and the shape fitted. * * What the shape misses is that swipes come in runs. Working through a * backlog is five gestures in five seconds, and it used to cost five writes * and five dashboard reloads — every one of them a multi-second execution * behind the others, each answer landing on a screen that had moved on. * Batching makes the same run one write and one reload if you swipe faster * than the round trip, which on this platform is most of the time. * * The optimistic half is untouched by any of this: a queued decision has * already redrawn its row, so waiting for the call to leave costs the screen * nothing. */ var queuedDecisions = []; var decisionsInFlight = false; function sendDecision(record, decision) { // Optimistic, and the local copies are what get redrawn — the server's own // view is refetched afterwards rather than assumed. var was = { confirmed: record.confirmed, category: record.category, subcategory: record.subcategory, labels: record.labels, type: record.type, inReports: record.inReports, amount: record.amount, rawAmount: record.rawAmount }; record.confirmed = true; // The row keeps its direction, so the new figure takes the old one's sign — // `editedAmount_`'s rule, predicted. Both fields, since a cash row is in the // base currency and `amount_base` is `amount` until the formula says so. if (decision.amount !== undefined) { var signed = Number(record.rawAmount) < 0 ? -decision.amount : decision.amount; record.rawAmount = signed; record.amount = signed; } if (decision.subcategory_override) record.subcategory = decision.subcategory_override; if (decision.category_override) { record.category = decision.category_override; // `type` is derived from the category on the sheet, so predicting it here // is what makes marking a transfer redraw as one immediately rather than // a round trip later. `in_reports` follows it, which is what stops a row // you just excluded from still colouring itself as spending. var kind = kindOfCategory(decision.category_override); if (kind) { record.type = kind; record.inReports = kind !== TRANSFER; } } if (decision.labels !== undefined) record.labels = decision.labels.slice(); // Noted before the call leaves, so a fetch that goes out while this is // queued or in flight cannot hand the row back — see `agreeWithLocalWrites`. noteWrite(record.uid, 'confirmed'); queuedDecisions.push({ record: record, decision: decision, was: was, dropped: dropConfirmed(record) }); render(); flushDecisions(); } /** * Put one refused row back the way it was, and say so. * * Returns the reason rather than toasting it: a batch can be refused whole — * a dropped connection refuses every row in it — and five banners in a row is * four the screen never gets to show. The caller says it once. * * `forgetWrite` is the half that is easy to miss. A row this page believes it * confirmed is a row it will go on hiding from every fetch; a refusal means it * never happened, so the note has to go with the tick. */ function undoDecision(entry, reason) { var record = entry.record; record.confirmed = entry.was.confirmed; record.category = entry.was.category; record.subcategory = entry.was.subcategory; record.labels = entry.was.labels; record.type = entry.was.type; record.inReports = entry.was.inReports; record.amount = entry.was.amount; record.rawAmount = entry.was.rawAmount; restoreDropped(record, entry.dropped); forgetWrite(record.uid); return reason; } /** * Send everything swiped since the last call left, as one call. * * The answer is matched back to the rows **by uid**, not by position: * `applyReviewDecisions_` reports one result per decision and is free to skip * one, and a batch aligned by index would then apply each row's answer to its * neighbour — a refusal restoring the wrong transaction, silently. * * **A row with no result at all is a refusal.** `MAX_APP_DECISIONS` caps a * batch server-side and reports `capped` rather than a result per row left * out, so a decision that fell off the end has no outcome to be `ok: false` — * and treating a missing answer as success is how a row ends up ticked on * screen, hidden from every refetch, and unconfirmed in the book. */ function flushDecisions() { if (decisionsInFlight || !queuedDecisions.length) return; var batch = queuedDecisions; queuedDecisions = []; decisionsInFlight = true; var settle = function () { decisionsInFlight = false; flushDecisions(); }; call('appConfirm', batch.map(function (entry) { return entry.decision; })) .then(function (result) { var byUid = {}; (result.results || []).forEach(function (outcome) { byUid[String(outcome.uid)] = outcome; }); var refused = []; var written = 0; batch.forEach(function (entry) { var outcome = byUid[String(entry.decision.uid)]; if (!outcome || !outcome.ok) { return refused.push(undoDecision(entry, outcome ? outcome.reason : 'the server did not answer for it')); } written++; writeLanded(entry.record.uid); /* `record` is the row as the sheet describes it *now* — the derived columns are formulas, so what a confirmation means is only knowable by asking, and the optimistic patch above is a guess at what it will say. */ if (outcome.record) patchRecord(entry.record, outcome.record); }); /* No toast on success, deliberately. The row has already redrawn — it carries a tick, it may have changed category, it may have left the list entirely if you were filtering on unconfirmed. A banner saying "Saved." adds nothing to that and costs four seconds of the screen, every single time, on the action you do most often. A refusal still speaks, because a refusal has nothing else to show for itself: the row goes back to how it was, which looks exactly like a tap that never landed. */ render(); if (refused.length) { toast(refused.length === 1 ? 'Not saved: ' + refused[0] : refused.length + ' not saved: ' + refused[0]); } if (written) reloadAfterWrite(); settle(); }) .catch(function (error) { batch.forEach(function (entry) { undoDecision(entry, error.message); }); toast(error.message); render(); settle(); }); }
