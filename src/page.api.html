// ------------------------------------------------------------------------- // Talking to the server // ------------------------------------------------------------------------- /** * `google.script.run`, as a promise. * * The callback form nests badly once a screen needs two things, and every * failure path here has to end in a message a person can read — a rejected * promise with one `catch` is how that stays true as screens are added. */ function call(name) { var args = Array.prototype.slice.call(arguments, 1); return new Promise(function (resolve, reject) { google.script.run .withSuccessHandler(resolve) .withFailureHandler(function (error) { reject(new Error(error && error.message ? error.message : String(error))); })[name].apply(null, args); }); } /** How long to wait before the one retry. Long enough for a lift to pass. */ var RETRY_MS = 1200; /** * Errors from the transport rather than from our code. * * `http connection failure 0` is the sandbox host's own wording for an XHR * that died — a radio drop, a tab the OS suspended, a session that expired, * or an execution that queued behind another for long enough that the bridge * gave up. None of those are answers to the question that was asked, and all * of them are reported through the same `withFailureHandler` as a genuine * refusal from `requireOwner_`. Matched on the wording because that is all * there is: the object carries no status and no type. */ function isTransportFailure(error) { var message = String((error && error.message) || error || '').toLowerCase(); return message.indexOf('connection failure') !== -1 || message.indexOf('network') !== -1 || message.indexOf('timed out') !== -1; } /** * `call`, with one retry when the transport dropped it. * * **One.** A retry loop against a server that is genuinely unwell turns one * failure into a queue of them, and this is a phone on a train: the second * attempt either works or the answer really is "not now". A refusal from our * own code is never retried, because asking twice cannot change it. * * The reason this earns its place is what the failures looked like. Every * write called `appDashboard`, which forgot the ledger cache and rebuilt it * from four and a half thousand rows — so confirming three rows in a row * queued three multi-second executions, and the bridge dropped the ones at * the back. That cause is fixed at the source; this is for the radio. */ function callOrRetry(name) { var args = arguments; return call.apply(null, args).catch(function (error) { if (!isTransportFailure(error)) throw error; return new Promise(function (resolve) { setTimeout(resolve, RETRY_MS); }) .then(function () { return call.apply(null, args); }); }); } /** The filters, as a string, so a result can be cached against them. */ function signature(filters) { return JSON.stringify([filters.period, filters.account, filters.label, filters.search || '', !!filters.unconfirmedOnly, !!filters.spendOnly, !!filters.incomeOnly, filters.category || 'All', filters.subcategory || 'All']); }
